%     acknowledge/  - 致谢文件夹，放入致谢.tex文件
%     pic/          - 图片文件夹，放入所有用到的图片
%
%   【文件对应关系】
%     main.tex      - 主文件，编译入口，不要删除开头的注释说明
%     cover/cover_cyber.tex  - 封面内容，修改这里调整封面样式
%     abstract/abstract.tex  - 中文摘要内容
%     body/1.tex    - 第一章：绪论
%     body/2.tex    - 第二章：相关技术介绍
%     body/3.tex    - 第三章：系统需求分析与总体设计
%     body/4.tex    - 第四章：A*寻路算法设计与优化
%     body/5.tex    - 第五章：系统详细设计与实现
%     body/6.tex    - 第六章：系统测试与结果分析
%     body/7.tex    - 第七章：总结与展望
%     acknowledge/acknowledge.tex  - 致谢内容
%
%   【编译方式】
%     必须使用XeLaTeX编译器编译，不能用PDFLaTeX或LuaLaTeX

%     命令行方式：xelatex main.tex（需要多次运行生成目录和引用）
%
%   【图片插入方法】
%     \includegraphics[width=0.8\textwidth]{pic/图片文件名}
%     图片文件放入pic/文件夹，使用相对路径引用
%     建议使用PNG或PDF格式，EPS格式可能需要不同处理
%
%   【文献引用方法】
%     在thebibliography环境中手动添加引用条目
%     引用格式：\bibitem[1]{ref1}作者. 题目[J]. 期刊, 年份.
%     文中引用：\upcite{1} 会在文中显示上标[1]
%
%   【常见转义字符】
%     \#  - 生成井号 #
%     \$  - 生成美元符号 $（用于数学公式内另当别论）
%     \%  - 生成百分号 %（用于注释另当别论）
%     \&  - 生成连字符 &
%     \_  - 生成下划线 _
%     \{  - 生成左花括号 {
%     \}  - 生成右花括号 }
%     \S  - 生成章节号 §
%     \P  - 生成段落号 ¶
%     \dag - 生成剑号 †
%     \ddag - 生成双剑号 ‡
%
%   【如何在正文输入%符号】
%     方法一：\verb|%|
%     方法二：\texttt{\%}
%     方法三（仅用于数学公式中）：\# 即可
%
%   【特殊符号】（需加载amssymb宏包）
%     \checkmark  - √ 对勾
%     \star      - ★ 五角星
%     \diamond   - ◆ 菱形
%     \blacksquare - ■ 黑方块
%     \triangle - △ 三角形
%     \circ     - ○ 空心圆
%     \bullet   - ● 实心圆点
%     \infty    - ∞ 无穷大
%
%   【代码块插入方法】
%     \begin{lstlisting}[language=python]
%     your code here
%     \end{lstlisting}
%     language参数可选：python, java, c, cpp, matlab等
%
%   【表格详细用法】
%
%     1. 最简单的三线表格式：
%        \begin{table}[H]
%          \centering
%          \caption{表格标题}
%          \begin{tabular}{ccc}
%            \hline
%            列1 & 列2 & 列3 \\
%            \hline
%            内容1 & 内容2 & 内容3 \\
%            内容4 & 内容5 & 内容6 \\
%            \hline
%          \end{tabular}
%        \end{table}
%
%     2. 表格参数说明：
%        {ccc} 表示三列，都是居中对齐
%        c = 居中，l = 左对齐，r = 右对齐
%        |c|c|c| 加竖线表示列之间的分隔线
%        \hline 表示横线
%
%     3. 带列标题的表格（使用booktabs宏包风格）：
%        \begin{table}[H]
%          \centering
%          \caption{实验数据}
%          \begin{tabular}{>{\raggedright}X >{\centering}X >{\centering}X}
%            \hline
%            \textbf{算法} & \textbf{准确率} & \textbf{耗时} \\
%            \hline
%            A*算法 & 95.2\% & 12ms \\
%            BFS算法 & 89.1\% & 25ms \\
%            \hline
%          \end{tabular}
%        \end{table}
%
%     4. 表格宽度自动适应页面（tabularx环境）：
%        \begin{table}[H]
%          \centering
%          \caption{自适应宽度表格}
%          \begin{tabularx}{\linewidth}{>{\raggedright}X >{\centering}X >{\centering}X}
%            \hline
%            \textbf{列1} & \textbf{列2} & \textbf{列3} \\
%            \hline
%            自动拉伸 & 自动拉伸 & 自动拉伸 \\
%            \hline
%          \end{tabularx}
%        \end{table}
%        注意：需要 \usepackage{tabularx} 宏包
%        X参数使该列自动扩展填满表格宽度
%
%     5. 单元格内换行（makecell命令）：
%        \makecell{第一行内容\\第二行内容}
%        需要 \usepackage{makecell}
%
%     6. 常见问题：
%        - 表格超出页面：减小width值或用 tabularx 的X参数
%        - 单元格内容偏上/偏下：用 >{\centering\arraybackslash}m{宽度} 代替 c
%        - 需要跨列：用 \multicolum{3}{c}{内容}
%        - 需要跨行：用 \multirow{2}{宽度}{内容}
%
%     7. 本模板已配置的表格相关宏包：
%        - array     提供列对齐参数 >{...}
%        - tabularx  提供 X 参数自动列宽
%        - makecell  提供 \makecell 单元格内换行
%        - booktabs  提供 \toprule \midrule \bottomrule 美观横线

%     行内公式用 $...$ 或 \(...\)
%     行间公式用 \[...\] 或 \begin{equation}...\end{equation}
