\subsection{核心数据结构实现}

\subsubsection{状态类设计}

状态类（\texttt{State}）是系统中最核心的数据结构，采用面向对象设计：

\begin{lstlisting}[language=c++,caption={State类定义（state.hpp）}]
class State {
public:
    std::vector<Position> boxes;  // 箱子位置集合
    Position player;              // 玩家位置
    int width;                    // 地图宽度
    int height;                   // 地图高度
    uint64_t hash;               // Zobrist哈希值

    State() : width(0), height(0), hash(0) {}

    State(int w, int h, const std::vector<Position>& boxes,
          const Position& player)
        : boxes(boxes), player(player), width(w), height(h), hash(0) {
        std::sort(this->boxes.begin(), this->boxes.end());
    }

    bool operator==(const State& other) const;
    bool operator<(const State& other) const;
    uint64_t computeHash(const std::vector<std::vector<bool>>& wallGrid,
                         const std::vector<uint64_t>& boxHash,
                         const std::vector<uint64_t>& playerHash) const;
    std::string toString() const;
};
\end{lstlisting}

状态类的设计要点：
\begin{itemize}
\item \texttt{boxes}使用向量存储并始终保持有序，确保相同状态有相同的向量表示
\item \texttt{hash}存储Zobrist哈希值，用于快速状态比较和去重
\item 拷贝构造函数默认实现，支持深拷贝
\end{itemize}

\subsubsection{位置结构设计}

\begin{lstlisting}[language=c++,caption={Position结构定义（state.hpp）}]
struct Position {
    int x, y;
    Position() : x(0), y(0) {}
    Position(int x_, int y_) : x(x_), y(y_) {}

    bool operator==(const Position& other) const {
        return x == other.x && y == other.y;
    }

    bool operator<(const Position& other) const {
        return y < other.y || (y == other.y && x < other.x);
    }

    Position move(int dir) const {
        return Position(x + DIR_X[dir], y + DIR_Y[dir]);
    }
};
\end{lstlisting}

位置结构重载了比较运算符，使其可以直接用于STL容器（如集合、优先队列）。

\subsection{Zobrist哈希实现}

Zobrist哈希的实现利用了64位无符号整数的异或运算性质：

\begin{lstlisting}[language=c++,caption={Zobrist哈希实现（zobrist.hpp）}]
class ZobristTable {
public:
    std::vector<std::vector<std::vector<uint64_t>>> boxTable;
    std::vector<std::vector<uint64_t>> playerTable;
    uint64_t currentHash;

    void init(int height, int width, int numBoxes) {
        boxTable.resize(height);
        for (int y = 0; y < height; y++) {
            boxTable[y].resize(width);
            for (int x = 0; x < width; x++) {
                boxTable[y][x].resize(numBoxes);
                for (int b = 0; b < numBoxes; b++) {
                    boxTable[y][x][b] = RandomGenerator::next();
                }
            }
        }

        playerTable.resize(height);
        for (int y = 0; y < height; y++) {
            playerTable[y].resize(width);
            for (int x = 0; x < width; x++) {
                playerTable[y][x] = RandomGenerator::next();
            }
        }
        currentHash = RandomGenerator::next();
    }

    uint64_t computeHash(const std::vector<Position>& boxes,
                         const Position& player) const {
        uint64_t h = currentHash;
        for (int i = 0; i < static_cast<int>(boxes.size()); i++) {
            h ^= boxTable[boxes[i].y][boxes[i].x][i];
        }
        h ^= playerTable[player.y][player.x];
        return h;
    }
};
\end{lstlisting}

\subsection{启发式函数实现}

\subsubsection{曼哈顿距离启发式}

曼哈顿距离启发式是最基础的实现：

\begin{lstlisting}[language=c++,caption={曼哈顿距离启发式（heuristics.cpp）}]
int ManhattanDistance::h(const State& state) const {
    int sum = 0;
    for (const auto& box : state.boxes) {
        int minDist = std::numeric_limits<int>::max();
        for (const auto& target : detector->targets) {
            int dist = std::abs(box.x - target.x) + std::abs(box.y - target.y);
            minDist = std::min(minDist, dist);
        }
        if (minDist == std::numeric_limits<int>::max()) {
            return std::numeric_limits<int>::max();
        }
        sum += minDist;
    }

    if (detector && detector->isDeadlock(state)) {
        return std::numeric_limits<int>::max();
    }

    return sum;
}
\end{lstlisting}

\subsubsection{最小权完美匹配启发式}

最小权完美匹配启发式使用匈牙利算法求解最优分配：

\begin{lstlisting}[language=c++,caption={匈牙利算法实现（heuristics.cpp）}]
int MinimumWeightPerfectMatching::hungarian(
    const std::vector<std::vector<int>>& cost,
    int n, std::vector<int>& match) const {

    std::vector<int> u(n+1), v(n+1), p(n+1), way(n+1);

    for (int i = 1; i <= n; i++) {
        p[0] = i;
        int j0 = 0;
        std::vector<int> minv(n+1, INT_MAX);
        std::vector<char> used(n+1, false);

        do {
            used[j0] = true;
            int i0 = p[j0], delta = INT_MAX, j1 = 0;

            for (int j = 1; j <= n; j++) {
                if (!used[j]) {
                    int cur = cost[i0-1][j-1] - u[i0] - v[j];
                    if (cur < minv[j]) {
                        minv[j] = cur;
                        way[j] = j0;
                    }
                    if (minv[j] < delta) {
                        delta = minv[j];
                        j1 = j;
                    }
                }
            }

            for (int j = 0; j <= n; j++) {
                if (used[j]) {
                    u[p[j]] += delta;
                    v[j] -= delta;
                } else {
                    minv[j] -= delta;
                }
            }

            j0 = j1;
        } while (p[j0] != 0);

        do {
            int j1 = way[j0];
            p[j0] = p[j1];
            j0 = j1;
        } while (j0);
    }

    match.assign(n, -1);
    for (int j = 1; j <= n; j++) {
        match[p[j]-1] = j-1;
    }

    return -v[0];
}
\end{lstlisting}

\subsection{死锁检测实现}

死锁检测是推箱子求解系统中的关键优化：

\begin{lstlisting}[language=c++,caption={角落死锁检测（state.cpp）}]
bool DeadlockDetector::isCornerDeadlock(const Position& box) const {
    int x = box.x;
    int y = box.y;

    if (targetGrid[y][x]) {
        return false;
    }

    bool leftWall = (x - 1 < 0) || wallGrid[y][x - 1];
    bool rightWall = (x + 1 >= width) || wallGrid[y][x + 1];
    bool topWall = (y - 1 < 0) || wallGrid[y - 1][x];
    bool bottomWall = (y + 1 >= height) || wallGrid[y + 1][x];

    if (leftWall && topWall) return true;
    if (rightWall && topWall) return true;
    if (leftWall && bottomWall) return true;
    if (rightWall && bottomWall) return true;

    return false;
}
\end{lstlisting}

\subsection{IDA*搜索实现}

IDA*是系统的核心搜索算法：

\begin{lstlisting}[language=c++,caption={IDA*搜索实现（ida_star.cpp）}]
int IDASolver::search(int g, int bound, State& state,
                       std::vector<State>& path,
                       std::vector<Direction>& moves) {
    int h = f(state, g);

    if (h == std::numeric_limits<int>::max()) {
        return std::numeric_limits<int>::max();
    }

    int f = g + h;
    if (f > bound) {
        return f;
    }

    if (h == 0) {
        return -1;
    }

    nodesExpanded++;
    maxDepthReached = std::max(maxDepthReached, g);

    if (isTimeExpired() || cancelRequested) {
        return std::numeric_limits<int>::max();
    }

    auto legalMoves = getLegalMoves(state);
    int minBound = std::numeric_limits<int>::max();

    for (const auto& [dir, boxPos] : legalMoves) {
        State newState = applyMove(state, dir, boxPos);

        auto it = visited.find(newState.hash);
        if (it != visited.end() && it->second <= g + 1) {
            continue;
        }

        visited[newState.hash] = g + 1;
        path.push_back(newState);
        moves.push_back(dir);

        int t = search(g + 1, bound, newState, path, moves);

        if (t == -1) {
            return -1;
        }

        if (t < minBound) {
            minBound = t;
        }

        path.pop_back();
        moves.pop_back();
    }

    return minBound == std::numeric_limits<int>::max() ? bound : minBound;
}
\end{lstlisting}

\subsection{地图解析器实现}

地图解析器支持标准XSB格式：

\begin{lstlisting}[language=c++,caption={XSB格式解析（map_parser.cpp）}]
MapParser::ParseResult MapParser::parseLines(
    const std::vector<std::string>& lines) {

    ParseResult result;

    size_t maxWidth = 0;
    for (const auto& l : lines) {
        maxWidth = std::max(maxWidth, l.size());
    }

    std::vector<std::string> normalized;
    for (const auto& l : lines) {
        if (!l.empty()) {
            std::string row = l;
            while (row.size() < maxWidth) row += ' ';
            normalized.push_back(row);
        }
    }

    for (int y = 0; y < static_cast<int>(normalized.size()); y++) {
        const std::string& l = normalized[y];
        for (int x = 0; x < static_cast<int>(l.size()); x++) {
            char c = l[x];
            Position pos(x, y);

            if (c == PLAYER || c == PLAYER_ON_TARGET) {
                result.player = pos;
            }
            if (c == BOX || c == BOX_ON_TARGET) {
                result.boxes.push_back(pos);
            }
            if (c == TARGET || c == BOX_ON_TARGET || c == PLAYER_ON_TARGET) {
                result.targets.push_back(pos);
            }
        }
    }

    result.map = normalized;
    result.height = static_cast<int>(normalized.size());
    result.width = static_cast<int>(maxWidth);
    result.success = true;
    return result;
}
\end{lstlisting}

\subsection{游戏引擎实现}

游戏引擎处理推箱子的基本规则：

\begin{lstlisting}[language=c++,caption={移动逻辑实现（game_engine.cpp）}]
MoveResult GameEngine::move(State& state, Direction dir) {
    Position newPlayerPos = state.player.move(dir);

    if (!inBounds(newPlayerPos.x, newPlayerPos.y)) {
        return MoveResult::OUT_OF_BOUNDS;
    }

    if (isWall(newPlayerPos.x, newPlayerPos.y)) {
        return MoveResult::HIT_WALL;
    }

    int boxIdx = -1;
    for (int i = 0; i < static_cast<int>(state.boxes.size()); i++) {
        if (state.boxes[i].x == newPlayerPos.x &&
            state.boxes[i].y == newPlayerPos.y) {
            boxIdx = i;
            break;
        }
    }

    if (boxIdx >= 0) {
        Position boxDest = newPlayerPos.move(dir);

        if (!inBounds(boxDest.x, boxDest.y)) {
            return MoveResult::OUT_OF_BOUNDS;
        }

        if (isWall(boxDest.x, boxDest.y)) {
            return MoveResult::HIT_WALL;
        }

        bool destHasBox = false;
        for (const auto& box : state.boxes) {
            if (box.x == boxDest.x && box.y == boxDest.y) {
                destHasBox = true;
                break;
            }
        }
        if (destHasBox) {
            return MoveResult::BOX_STUCK;
        }

        state.boxes[boxIdx] = boxDest;
        std::sort(state.boxes.begin(), state.boxes.end());
    }

    state.player = newPlayerPos;
    return MoveResult::OK;
}
\end{lstlisting}
